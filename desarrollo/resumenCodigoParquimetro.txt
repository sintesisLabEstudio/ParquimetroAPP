================================================================================
RESUMEN TÉCNICO Y FUNCIONAL: SISTEMA DE PARQUÍMETRO ATLETICOCON NUMERO
================================================================================

FUNCIONALIDADES:
- Cálculo automatizado de tarifa de parquímetro basado en tiempo real de estacionamiento (Horas y Minutos).
- Conversión inteligente de tiempo a una unidad base única (minutos) para aplicar lógica de negocio.
- Aplicación de tramos de cobro:
    - Tramo Base: Cobro fijo de $2850 para estacionamientos de 1 a 25 minutos.
    - Tramo Excedente: Cobro de $2850 más un adicional de $28 por cada minuto que supere los 25 minutos iniciales.
- Interfaz de usuario intuitiva con campos de entrada numérica específicos para horas y minutos.
- Sistema de retroalimentación mediante Toasts (mensajes emergentes) que informan el tiempo total en minutos y el monto final.
- Visualización permanente del monto a cancelar en pantalla.
- Diseño adaptativo para dispositivos móviles con soporte para barras de sistema dinámicas (Edge-to-Edge).

package com.example.apppuntosdecimas

import android.os.Bundle
import android.widget.Button
import android.widget.EditText
import android.widget.TextView
import android.widget.Toast
import androidx.activity.enableEdgeToEdge
import androidx.appcompat.app.AppCompatActivity
import androidx.core.view.ViewCompat
import androidx.core.view.WindowInsetsCompat

class MainActivity : AppCompatActivity() {
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)

        // Habilita el diseño de borde a borde para usar toda la superficie de la pantalla
        enableEdgeToEdge()

        // Asocia esta clase con su diseño visual definido en el archivo XML
        setContentView(R.layout.activity_main)

        // Ajusta los márgenes para que los elementos no queden tapados por la muesca o barras del sistema
        ViewCompat.setOnApplyWindowInsetsListener(findViewById(R.id.main)) { v, insets ->
            val systemBars = insets.getInsets(WindowInsetsCompat.Type.systemBars())
            v.setPadding(systemBars.left, systemBars.top, systemBars.right, systemBars.bottom)
            insets
        }

        // --- DEFINICIÓN DE CONSTANTES DEL NEGOCIO ---
        // Estas variables guardan los precios y tiempos definidos por la administración del parquímetro
        // val tiempoEst = (horasEst/60) + minutosEst
        val tarifaBase = 2850      // Costo fijo inicial
        val maxMinutos = 25        // Límite del tramo base
        val minAdd = 28            // Precio del minuto adicional
        var pagoTotal = 0          // Variable acumuladora para el resultado final

        // --- VINCULACIÓN DE COMPONENTES (ID a Variables) ---
        // Conectamos las variables de código con los objetos definidos en el XML mediante sus IDs
        val txtInputHoras = findViewById<EditText>(R.id.txtInputHoras)
        val txtInputMinutos = findViewById<EditText>(R.id.txtInputMinutos)
        val btnCalcula = findViewById<Button>(R.id.btnCalcula)
        val outPago = findViewById<TextView>(R.id.txvOutPago)

        // --- LÓGICA DEL BOTÓN CALCULAR ---
        // Este bloque se ejecuta cada vez que el usuario presiona el botón "Calcular"
        btnCalcula.setOnClickListener {

            // --- FASE 1: CAPTURA DE DATOS ---
            // Leemos lo que el usuario escribió en los campos de Horas y Minutos
            val txtHoras = txtInputHoras.text.toString()
            val txtMinutos = txtInputMinutos.text.toString()

            // Notificamos al usuario los datos capturados para transparencia
            Toast.makeText(this, "Horas: $txtHoras, Minutos: $txtMinutos", Toast.LENGTH_SHORT).show()

            // --- FASE 2: PROCESAMIENTO (CÁLCULO) ---
            // Convertimos el texto a número. Si está vacío, se asume 0 para evitar errores.
            val hora = txtHoras.toIntOrNull() ?: 0
            val minuto = txtMinutos.toIntOrNull() ?: 0

            // Convertimos las horas a minutos y sumamos los minutos restantes
            val tiempoMinutos = (hora * 60) + minuto
            val txtTiempoMin = tiempoMinutos.toString()

            // --- FASE 3: REGLAS DEL NEGOCIO (ALGORITMO DE COBRO) ---
            if (tiempoMinutos > 0 && tiempoMinutos <= maxMinutos) {
                // Si el tiempo es menor o igual a 25 minutos, se cobra solo la base
                pagoTotal = tarifaBase
            } else {
                // Si excede los 25 minutos, se suma el costo adicional por cada minuto extra
                pagoTotal = tarifaBase + (minAdd * (tiempoMinutos - maxMinutos))
            }

            // Formateamos el resultado final para ser mostrado
            val txtPagar = pagoTotal.toString()
            val mensaje = "DEBE CANCELAR: $" + txtPagar

            // --- FASE 4: PRESENTACIÓN DE RESULTADOS ---
            // Mostramos el tiempo total calculado en un mensaje flotante
            Toast.makeText(this, "tiempo total en minutos: $txtTiempoMin", Toast.LENGTH_LONG).show()

            // Mostramos el monto a pagar en un mensaje flotante
            Toast.makeText(this, mensaje, Toast.LENGTH_LONG).show()

            // Actualizamos la etiqueta en pantalla con el resultado final
            outPago.text = mensaje
        }
    }
}
